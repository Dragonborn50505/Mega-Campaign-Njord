estate_peasants_allow_peasant_freedoms = {
	icon = privilege_recruit_minister
	land_share = 0
	max_absolutism = -10
	conditional_modifier = { trigger = { has_absolutism_reduction_for_estate_privileges = yes NOT = { has_government_attribute = free_noble_levies } } modifier = { max_absolutism = 2 } }
	loyalty = 0.1
	influence = 0.1
	is_valid = {
		num_of_owned_provinces_with = {
			value = 5
			OR = {
				trade_goods = fish
				trade_goods = grain
				trade_goods = livestock
			}
		}
	}
	on_granted = {
		custom_tooltip = BM_estate_peasants_guild_assembly_grant_tooltip
		tooltip = {
			add_province_modifier = {
				name = BM_estate_peasants_guild_assemblies
				duration = -1
			}
		}
		hidden_effect = {
			every_owned_province = {
			limit = {
				OR = {
					trade_goods = fish
					trade_goods = grain
					trade_goods = livestock
				}
			}
			add_province_modifier = {
				name = BM_estate_peasants_guild_assemblies
				duration = -1
			}
		}
		}
	}
	on_revoked = {
		custom_tooltip = BM_estate_peasants_guild_assembly_revoke_tooltip
		tooltip = {
			add_province_modifier = {
				name = BM_estate_peasants_guild_assemblies
				duration = -1
			}
		}
	}
	on_revoked_province = {
		if = {
			limit = {
				has_province_modifier = BM_estate_peasants_guild_assemblies
			}
			remove_province_modifier = BM_estate_peasants_guild_assemblies
		}
	}
	on_invalid = {
		custom_tooltip = BM_estate_peasants_guild_assembly_revoke_tooltip
	}
	on_invalid_province = {
		if = {
			limit = {
				has_province_modifier = BM_estate_peasants_guild_assemblies
			}
			remove_province_modifier = BM_estate_peasants_guild_assemblies
		}
	}
	can_select = {
		num_of_owned_provinces_with = {
			value = 5
			OR = {
				trade_goods = fish
				trade_goods = grain
				trade_goods = livestock
			}
		}
	}
	penalties = {
	}
	benefits = {
	}
	ai_will_do = {
		factor = 0
	}
}
